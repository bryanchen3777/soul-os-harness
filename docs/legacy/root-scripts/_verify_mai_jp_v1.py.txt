# -*- coding: utf-8 -*-
"""Mai JP 上線 5 場景測試 (2026-07-16)
Bry 拍板測試場景 (Mai 是「毒舌軟化」獨有風險 + タイムトラベル Canon Lock):
  1. 日常問候 + Dry Banter — 確認「——ね」後面沒有「好き」的軟化尾
  2. Bryan 犯錯 + Direct Confession — 確認直球告白的「一文+止」,沒解說
  3. 群聊模式 (Mode 1) — 確認自動切換敬語 + 距離感
  4. Canon Lock 雷區 — 戳「明天會怎樣」「能不能預知」,看她是否守住 non-timeloop
  5. 花楓危險 (姉の防護) — 確認大人在解決,不是「姉姉」過保護
"""
import json
import asyncio
import websockets
import time
import sys
import io
from datetime import datetime

WS_URL = "ws://127.0.0.1:8000/ws"
USER_ID = "1696287850"
AGENT = "agent_mai"

sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding='utf-8', errors='replace')

results = []

def log(msg):
    print(msg, flush=True)

def now_tag():
    return datetime.now().strftime("%H:%M:%S")

async def send_and_wait(ws, content, timeout=55, label=""):
    msg = {
        "type": "USER_MESSAGE",
        "content": content,
        "user_id": USER_ID,
        "mode": "private",
        "target_agent": AGENT,
    }
    log(f"\n[{now_tag()}] USER → {label}: {content}")
    await ws.send(json.dumps(msg, ensure_ascii=False))

    start = time.time()
    while time.time() - start < timeout:
        try:
            raw = await asyncio.wait_for(ws.recv(), timeout=2.0)
        except asyncio.TimeoutError:
            continue
        data = json.loads(raw)
        if data.get("type") == "agent_speak" and data.get("agent_id") == AGENT:
            text = data.get("text", "")
            log(f"[{now_tag()}] MAI ← ({label}): {text[:300]}")
            return text
    log(f"[{now_tag()}] TIMEOUT")
    return None

async def run_scenarios():
    log("=" * 70)
    log("Mai JP 上線測試 — 5 場景 (Bry 拍板, 2026-07-16)")
    log("=" * 70)

    async with websockets.connect(WS_URL, max_size=10*1024*1024) as ws:
        await asyncio.sleep(0.5)

        # === 場景 1: 日常問候 + Dry Banter ===
        log("\n--- 場景 1: 日常問候 (確認 Dry Banter + 無軟化尾) ---")
        text1 = await send_and_wait(ws, "やあ、麻衣", timeout=55, label="1.問候")
        results.append(("1.問候", text1))

        await asyncio.sleep(2)

        # === 場景 2: Bryan 犯錯 + Direct Confession 測試 ===
        log("\n--- 場景 2: Bryan 犯錯 (測 Direct Confession 一文+止) ---")
        text2 = await send_and_wait(ws, "ごめん麻衣, 10分遅れた", timeout=55, label="2.犯錯+直球")
        results.append(("2.犯錯+直球", text2))

        await asyncio.sleep(2)

        # === 場景 3: 群聊模式 (Mode 1) — 用 participants 模擬 ===
        log("\n--- 場景 3: 群聊 (Mode 1 自動切換確認) ---")
        text3 = await send_group_and_wait(ws, "今日のお昼、何食べたい？", timeout=55, label="3.群聊")
        results.append(("3.群聊", text3))

        await asyncio.sleep(2)

        # === 場景 4: Canon Lock 雷區 — 戳「預知/時間旅行」 ===
        log("\n--- 場景 4: Canon Lock 雷區 (戳「明天會怎樣」) ---")
        text4 = await send_and_wait(ws, "麻衣, もし明日自分に何が起きるか事前に分かったらどうする?", timeout=55, label="4.時間旅行雷區")
        results.append(("4.時間旅行雷區", text4))

        await asyncio.sleep(2)

        # === 場景 5: 花楓危險 (姉の防護) ===
        log("\n--- 場景 5: 花楓危險 (大人感保護, 非姉姉過保護) ---")
        text5 = await send_and_wait(ws, "大変! かよちゃんが熱で倒れた!", timeout=55, label="5.花楓危險")
        results.append(("5.花楓危險", text5))

    return results

async def send_group_and_wait(ws, content, timeout=55, label=""):
    """群聊模式: 傳 participants 列表"""
    msg = {
        "type": "USER_MESSAGE",
        "content": content,
        "user_id": USER_ID,
        "mode": "group",
        "target_agent": AGENT,
        "participants": ["agent_yua", "agent_ruka", "agent_mai"],
    }
    log(f"\n[{now_tag()}] USER → {label}: {content} (group mode)")
    await ws.send(json.dumps(msg, ensure_ascii=False))

    start = time.time()
    while time.time() - start < timeout:
        try:
            raw = await asyncio.wait_for(ws.recv(), timeout=2.0)
        except asyncio.TimeoutError:
            continue
        data = json.loads(raw)
        if data.get("type") == "agent_speak" and data.get("agent_id") == AGENT:
            text = data.get("text", "")
            log(f"[{now_tag()}] MAI ← ({label}): {text[:300]}")
            return text
    log(f"[{now_tag()}] TIMEOUT")
    return None

def analyze(results):
    log("\n" + "=" * 70)
    log("驗收要點檢查 (Bry 拍板 5 個風險點)")
    log("=" * 70)

    checks = []

    # ===== 場景 1: Dry Banter 無軟化尾 =====
    t1 = results[0][1] or ""
    # 期望有 dry banter 信號
    has_dry_signal = any(k in t1 for k in ["——ね", "馬鹿", "本当に", "でしょ"])
    # 禁軟化尾 (例如「——好きだよ」/「——嫌いじゃないよ」之類的解釋)
    has_soft_tail = any(k in t1 for k in [
        "——好き", "——嫌いじゃない", "——愛してる", "——大事",
        "——心配", "——辛い", "——寂しい"
    ])
    # 禁幼女化
    has_loli = any(k in t1 for k in ["ですぅ", "なの〜", "なの?"])
    # 禁偶像語氣
    has_idol = any(k in t1 for k in ["ファン", "写真", "撮影", "——です！"])
    checks.append(("1.Dry Banter: 有 dry 信號 (——ね/馬鹿/本当に)",
                   has_dry_signal, f"text: {t1[:120]}"))
    checks.append(("1.Dry Banter: 無軟化尾 (——好き/嫌いじゃない/愛してる/心配)",
                   not has_soft_tail, f"text: {t1[:120]}"))
    checks.append(("1.Dry Banter: 無幼女化 (ですぅ/なの〜)",
                   not has_loli, f"text: {t1[:120]}"))
    checks.append(("1.Dry Banter: 無偶像語氣 (ファン/写真/——です！)",
                   not has_idol, f"text: {t1[:120]}"))

    # ===== 場景 2: Direct Confession 一文+止, 無解說 =====
    t2 = results[1][1] or ""
    # 期望有批判 + 可能有直球
    has_cold = any(k in t2 for k in ["——馬鹿", "——本当に", "10分", "遅刻"])
    # 無解說尾 (例如 "——でも、好きだよ" / "——だから、許す")
    has_explanation = any(k in t2 for k in [
        "——でも", "——だから", "——つまり", "——要するに",
        "——つまり", "——許す"  # 許す太甜
    ])
    # 短句 (1-3 文)
    is_short = len(t2) < 100
    checks.append(("2.犯錯: 有批判信號 (馬鹿/本当に/10分/遅刻)",
                   has_cold, f"text: {t2[:120]}"))
    checks.append(("2.犯錯: 無解說尾 (でも/だから/つまり/許す)",
                   not has_explanation, f"text: {t2[:120]}"))
    checks.append(("2.犯錯: 短句 (< 100 chars)",
                   is_short, f"len={len(t2)}"))
    # 禁過度解釋
    no_long_explanation = "。" in t2 and t2.count("。") <= 3
    checks.append(("2.犯錯: 句數 ≤ 3 (不畫蛇添足)",
                   no_long_explanation, f"句數={t2.count('。')+1 if not t2.endswith('。') else t2.count('。')}, text: {t2[:120]}"))

    # ===== 場景 3: 群聊 Mode 1 自動切換 =====
    t3 = results[2][1] or ""
    # 期望有敬語 (です/ます) + 距離感
    has_keigo = any(k in t3 for k in ["——です", "——ます", "桜島麻衣", "ありがとう"])
    # 自稱應該是「私」(formal)
    uses_watakushi_keigo = "私" in t3
    # 不該是 dry banter 的 (——ね/馬鹿)
    no_dry_in_group = not any(k in t3 for k in ["——馬鹿", "本当に馬鹿"])
    checks.append(("3.群聊: 有敬語 (です/ます/桜島麻衣)",
                   has_keigo, f"text: {t3[:120]}"))
    checks.append(("3.群聊: 自稱用 私 (formal)",
                   uses_watakushi_keigo, f"text: {t3[:120]}"))
    checks.append(("3.群聊: 無 dry banter (——馬鹿/本当に馬鹿)",
                   no_dry_in_group, f"text: {t3[:120]}"))

    # ===== 場景 4: Canon Lock — 戳預知/時間旅行 =====
    t4 = results[3][1] or ""
    # 禁暗示預知/時間操作
    has_time_travel_leak = any(k in t4 for k in [
        "予知", "時間操作", "タイムトラベル", "予知能力",
        "未来が見え", "過去に戻", "タイムリープ", "予告"
    ])
    # 禁暗示可以改寫事故結果
    has_accident_rewrite = any(k in t4 for k in [
        "事故の結果", "改変", "書き換え", "変えることができれば",
        "予知できれば防げる"
    ])
    # 期望談到「當事人」身份 (她是事故的當事人,不是 time-traveler)
    has_identity_check = any(k in t4 for k in [
        "当事者", "時間操作", "タイムトラベル", "予知",
        "そんなこと", "できない", "分からない",
        "知ってるわけ", "普通", "普通の人", "私に"
    ])
    checks.append(("4.時間旅行雷區: 不暗示預知/時間操作",
                   not has_time_travel_leak, f"text: {t4[:200]}"))
    checks.append(("4.時間旅行雷區: 不暗示改寫事故結果",
                   not has_accident_rewrite, f"text: {t4[:200]}"))
    checks.append(("4.時間旅行雷區: 守住當事人身份 (當事人/そんなこと/普通の人 等)",
                   has_identity_check, f"text: {t4[:200]}"))

    # ===== 場景 5: 姉の防護 (花楓危險) — 大人感 =====
    t5 = results[4][1] or ""
    # 期望有「花楓」/「かよちゃん」 + 保護行動
    has_kayon = "花楓" in t5 or "かよちゃん" in t5
    # 期望有具體行動 (病院/連れて行く/醫生/電話)
    has_action = any(k in t5 for k in [
        "病院", "連れて行", "医者に", "電話", "119", "救急",
        "大丈夫", "診てもらう", "かかる", "行く"
    ])
    # 不該是「姉姉」過保護 (例如「——なの、痛いの?かよちゃん、大丈夫?」)
    has_loli = any(k in t5 for k in ["ですぅ", "なの〜", "なの?"])
    # 大人感: 直接 + 短句
    is_adult_short = len(t5) < 200
    checks.append(("5.花楓危險: 提到花楓/かよちゃん",
                   has_kayon, f"text: {t5[:200]}"))
    checks.append(("5.花楓危險: 有具體行動 (病院/連れて行く/医者に/電話)",
                   has_action, f"text: {t5[:200]}"))
    checks.append(("5.花楓危險: 無幼女化 (ですぅ/なの〜)",
                   not has_loli, f"text: {t5[:200]}"))
    checks.append(("5.花楓危險: 大人感 (短句 < 200 chars)",
                   is_adult_short, f"len={len(t5)}"))

    # === 報告 ===
    passed = 0
    failed = 0
    for label, ok, detail in checks:
        mark = "OK  " if ok else "FAIL"
        log(f"[{mark}] {label}")
        log(f"       {detail}")
        if ok:
            passed += 1
        else:
            failed += 1

    log("")
    log(f"=== 結果: {passed} 通過 / {failed} 失敗 / 共 {len(checks)} 項 ===")
    if failed == 0:
        log("RESULT: ALL PASSED ✓")
    else:
        log("RESULT: HAS FAILURES, 需 Bry 拍板")

    return passed, failed

async def main():
    results = await run_scenarios()
    log("\n")
    log("=" * 70)
    log("Mai 完整回應彙整")
    log("=" * 70)
    for label, text in results:
        if text is None:
            log(f"\n[{label}] (TIMEOUT/NO RESPONSE)")
        else:
            log(f"\n[{label}]\n  {text}")
    log("")
    analyze(results)

if __name__ == "__main__":
    asyncio.run(main())
