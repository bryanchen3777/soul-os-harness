try:
    import websockets
except ImportError:
    print("   ✗ websockets 沒裝, 跳過 WS probe")
    sys.exit(0)

async def probe():
    uri = "ws://127.0.0.1:8000/ws"
    async with websockets.connect(uri) as ws:
        # 發中文 user message 給 mahiru
        msg = {
            "type": "USER_MESSAGE",
            "content": "你在幹嘛",
            "user_id": "1696287850",
            "mode": "private",
            "target_agent": "agent_mahiru"
        }
        await ws.send(json.dumps(msg, ensure_ascii=False))
        print(f"   sent: {msg}")

        # 等 25s 收集回應
        responses = []
        try:
            for _ in range(30):
                resp_raw = await asyncio.wait_for(ws.recv(), timeout=1.0)
                resp = json.loads(resp_raw)
                responses.append(resp)
                if resp.get("type") == "agent_speak":
                    break
        except asyncio.TimeoutError:
            pass

        # 找 agent_speak
        speak = next((r for r in responses if r.get("type") == "agent_speak"), None)
        if not speak:
            print("   ✗ 沒收到 agent_speak, 看看其他 events:")
            for r in responses[:5]:
                print(f"     - {r.get('type')}: {str(r)[:200]}")
            return False

        text = speak.get("text", "")
        print(f"   text = {text!r}")

        # 驗證 hotfix #13 改動生效
        if "(中文翻譯生成失敗" in text:
            print("   ✗ FAIL: 還在加「(中文翻譯生成失敗...」註解")
            return False
        if "(LLM 回應解析失敗,請重試)" in text:
            print("   ✗ FAIL: 還在吐「(LLM 回應解析失敗,請重試)」")
            return False
        print("   ✓ text 不含任何「失敗」系統提示")

        # 確認 text 是日文 (有假名) 或空 (silent failure)
        has_kana = any('\u3040' <= c <= '\u309f' or '\u30a0' <= c <= '\u30ff' for c in text)
        if not has_kana and text:
            print(f"   ⚠ text 不是日文 ({text!r}), 可能是 LLM 吐的中文")
            return False
        if has_kana:
            print("   ✓ text 是日文 (有假名)")
        elif not text:
            print("   ✓ text 是 silent failure 空字串")
        return True

ok = asyncio.run(probe())
print()

sys.exit(0 if ok else 1)
