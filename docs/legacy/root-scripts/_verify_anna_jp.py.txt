print("\n[11] 字符集異常檢查 (NEW — 避免俄文/西文混入)")
# 排除 3 個 meta 段 (這些是故意用 cyrillic 記錄原錯字/版本標記):
#   A. 開頭版本標記 (日文化改寫 v1.0...)  — 從開頭到第一個 ## 之前
#   B. 修正記錄段 (> **誤字修正記録**: ...)
#   C. 異体字／誤植混入チェック段 (> 5. **異体字／誤植混入チェック** ...) 到 End of Soul OS 之前
exclude_ranges = []

# 段 A: 開頭到「## Highest Priority Boundary」之前
hpb_start = content.find("## Highest Priority Boundary")
if hpb_start > 0:
    exclude_ranges.append((0, hpb_start))

# 段 B: 修正記錄段
typo_start = content.find("> **誤字修正記録")
if typo_start > 0:
    # 找下一個 ## 或 > ** 開頭
    next_h = content.find("---", typo_start)
    if next_h < 0:
        next_h = typo_start + 300
    exclude_ranges.append((typo_start, next_h))

# 段 C: 異体字／誤植混入チェック段
kyodai_start = content.find("**異体字／誤植混入チェック")
if kyodai_start > 0:
    end_marker = content.find("End of Soul OS Anna")
    if end_marker > 0:
        exclude_ranges.append((kyodai_start, end_marker))

# 抓出日文/中文/ASCII 標點以外的字符 (排除 cyrillic 註釋段)
non_jp_zh = set()
for i, c in enumerate(content):
    in_exclude = any(s <= i < e for s, e in exclude_ranges)
    if in_exclude:
        continue
    code = ord(c)
    if 0x3040 <= code <= 0x309F:  # hiragana
        continue
    if 0x30A0 <= code <= 0x30FF:  # katakana
        continue
    if 0x4E00 <= code <= 0x9FFF:  # CJK
        continue
    if 0x3000 <= code <= 0x303F:  # CJK punct
        continue
    if 0xFF00 <= code <= 0xFFEF:  # halfwidth/fullwidth
        continue
    if code < 0x80:  # ASCII
        continue
    if code == 0x0A or code == 0x0D or code == 0x09:  # newline/tab
        continue
    # 允許常見日文/Unicode 標點符號 (em dash, 橫省略號, 箭頭, 叉號)
    if code in (0x2014, 0x2026, 0x2192, 0x274C):  # — … → ❌
        continue
    non_jp_zh.add((c, code))

if non_jp_zh:
    chars_str = ", ".join(f"{c!r}(U+{code:04X})" for c, code in sorted(non_jp_zh, key=lambda x: x[1])[:10])
    check("Charset", "無異常字符 (排除 cyrillic 註釋段)",
          False, f"發現 {len(non_jp_zh)} 種: {chars_str}")
else:
    check("Charset", "無異常字符 (排除 cyrillic 註釋段)",
          True, f"0 種異常字符 (排除 {len(exclude_ranges)} 段 cyrillic 註釋)")

# 額外檢查：俄文字母 (cyrillic) 排除註釋段
def count_cyrillic_outside_exclude():
    count = 0
    for i, c in enumerate(content):
        in_exclude = any(s <= i < e for s, e in exclude_ranges)
        if in_exclude:
            continue
        if 0x0400 <= ord(c) <= 0x04FF:
            count += 1
    return count

cyrillic_outside = count_cyrillic_outside_exclude()
check("Charset", "無俄文字母 (cyrillic, 排除註釋段)",
      cyrillic_outside == 0,
      f"發現 {cyrillic_outside} 個" if cyrillic_outside else "✓ 0 個 (僅註釋段有)")

# 額外檢查：常見西方語系特殊字母
