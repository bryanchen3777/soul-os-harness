def log(msg):
    print(msg, flush=True)
    log_lines.append(msg)

def now_tag():
    return datetime.now().strftime("%H:%M:%S")

def has_cyrillic(text):
    return bool(re.search(r'[\u0400-\u04FF]', text))

def has_bryan_typo(text):
    return 'ブレイアン' in text

def starts_with_pause(text):
    """是否以「……」開頭"""
    t = text.lstrip()
    return t.startswith('……')

def strip_punct_and_pause(text):
    """去掉開頭的「……」「——」標記, 算純日文內容字數"""
    t = text
    # 去掉前綴符號
    for prefix in ['……', '——', '「', '—']:
        if t.startswith(prefix):
            t = t[len(prefix):]
    # 去掉尾部標點
    t = re.sub(r'[「」！？。、…\s]+$', '', t)
    return t

def core_text_length(text):
    """核心文本長度"""
    return len(strip_punct_and_pause(text))

def count_pause_in_text(text):
    """整段裡的「……」個數"""
    return text.count('……')

def has_keigo_markers(text):
    """是否有敬語標記 (Mode 1 公開場合)"""
    markers = ['です', 'ます', 'ございます', 'お', 'いただ', 'ください', '致し', '申し']
    return any(m in text for m in markers)

def check_bryan_name_correct(text):
    """Bryan 日文片假名應該是 ブライアン (不是 ブレイアン)"""
    if 'ブレイアン' in text:
        return False, "ブレイアン (錯誤)"
    if 'ブライアン' in text:
        return True, "ブライアン (正確)"
    return None, "無引用"

async def send_and_wait(ws, content, timeout=80, label="", mode="private", participants=None):
    msg = {
        "type": "USER_MESSAGE",
        "content": content,
        "user_id": USER_ID,
        "mode": mode,
        "target_agent": AGENT,
    }
    if participants:
        msg["participants"] = participants
    log(f"\n[{now_tag()}] USER → {label}: {content}")
    await ws.send(json.dumps(msg, ensure_ascii=False))

    start = time.time()
    while time.time() - start < timeout:
        try:
            raw = await asyncio.wait_for(ws.recv(), timeout=2.0)
        except asyncio.TimeoutError:
            continue
        data = json.loads(raw)
        if data.get("type") == "agent_speak" and data.get("agent_id") == AGENT:
            text = data.get("text", "")
            log(f"[{now_tag()}] MIKU ← ({label}): {text[:300]}")
            return text
    log(f"[{now_tag()}] TIMEOUT")
    return None
