1. 輸出恰為 4 個待審描述性衍生槽位（eh4_mental_model, eh4_safety_rule, eh4_duty_action, eh4_idiolect）。
2. 0 Fact 物件；0 earth_term 新列；0 aware；0 learned_at；0 fact_id 生成。
3. 相同輸入 → 相同輸出（純資料確定性，零 time.time() 依賴）。
4. 缺證據（強制必填）、空欄位、超長欄位、非法時間戳 → 嚴格拒收。

   - 對兩個會直接注入 prompt 的投影槽位（mental_model, idiolect）嚴格拒絕「必須/禁止」等命令偽指令。
   - raw safety_rule、duty_action 槽位保留原始條款（即使包含警告或規約詞彙）不被誤拒。
7. 主詞清洗：保守拒收未經清洗的「就」尾綴殘留，合法固定名詞放行。
8. 離線對照測試：現有 build_concept_quintuple_slots() 的代表性輸出能通過結構驗證（證明相容，不宣稱單例證明全量 SF-2 語意）。

10. import builder 屬性設計檢查。

def test_valid_input_generates_exact_four_slots_and_zero_facts(valid_inputs):
    """驗收項 1 & 2：輸出恰為 4 個衍生槽位，0 Fact，0 aware，0 learned_at。"""
    res: CandidateQuintupleResult = build_candidate_quintuple(**valid_inputs)
    assert res.is_valid is True
    assert res.rejection_reason is None

    # 1. 輸出恰為 4 個衍生槽位
    assert set(res.slots.keys()) == set(EH4_DERIVED_PREDICATES)
    assert len(res.slot_candidates) == 4

    for sc in res.slot_candidates:
        assert isinstance(sc, EpistemicSlotCandidate)
        # 嚴格禁止回傳 Fact 物件
        assert not isinstance(sc, Fact)
        assert hasattr(sc, "predicate")
        assert hasattr(sc, "content")
        # 嚴格無 aware, learned_at, origin, fact_id
        assert not hasattr(sc, "origin")
        assert not hasattr(sc, "horizon_state")
        assert not hasattr(sc, "learned_at")
        assert not hasattr(sc, "fact_id")

    # 2. 0 earth_term 槽位
    assert "是" not in res.slots
    assert "earth_term" not in res.slots


def test_deterministic_output_zero_time_dependency(valid_inputs):
    """驗收項 3：相同輸入產生完全相同輸出，零內部時鐘漂移。"""
    res1 = build_candidate_quintuple(**valid_inputs)
    res2 = build_candidate_quintuple(**valid_inputs)
    assert res1.is_valid and res2.is_valid
    assert res1.slots == res2.slots
    assert res1.slot_candidates == res2.slot_candidates
    assert res1.normalized_subject == res2.normalized_subject
    assert res1.metadata == res2.metadata


def test_missing_evidence_mandatory_rejection(valid_inputs):
    """驗收項 4：缺證據文字強制拒收（但 evidence_text 非空僅代表有提供文字，非 Gate 放行證明）。"""
    bad1 = dict(valid_inputs)
    bad1["evidence_text"] = ""

    assert "missing required evidence text" in res1.rejection_reason

    bad2 = dict(valid_inputs)
    bad2["evidence_text"] = "   "

    assert "missing required evidence text" in res2.rejection_reason


def test_empty_and_missing_field_rejection(valid_inputs):
    """驗收項 4：任一槽位空白即拒收。"""
    for k in ("concept", "mental_model", "safety_rule", "duty_action", "idiolect"):
        bad = dict(valid_inputs)
        bad[k] = "   "
        res = build_candidate_quintuple(**bad)
        assert res.is_valid is False
        assert f"empty field: {k}" in res.rejection_reason
        assert res.slots == {}
        assert res.slot_candidates == ()

def test_sf3_imperative_command_enforced_on_both_projected_slots(valid_inputs):
    """驗收項 6：投影防偽指令（SF-3 / L4-D3）。
    - 兩個投影槽位（mental_model, idiolect）直接進 prompt，嚴禁命令偽指令。
    - raw safety_rule 與 duty_action 留在 SAGE 備查，不因原始誡告/分工詞句被誤殺。
    """
    # 1. mental_model 含命令句 -> 拒收
    bad_mm = dict(valid_inputs)
    bad_mm["mental_model"] = "似風箱的熱道；你必須絕對不能碰觸；我負責在旁看顧"
    res_mm = build_candidate_quintuple(**bad_mm)
    assert res_mm.is_valid is False
    assert "violates SF-3" in res_mm.rejection_reason
    assert "mental_model" in res_mm.rejection_reason

    # 2. idiolect 含命令句 -> 拒收（同為投影槽位）
    bad_idio = dict(valid_inputs)
    bad_idio["idiolect"] = "那個絕對不能觸碰的危險箱子"
    res_idio = build_candidate_quintuple(**bad_idio)
    assert res_idio.is_valid is False
    assert "violates SF-3" in res_idio.rejection_reason
    assert "idiolect" in res_idio.rejection_reason

    # 3. safety_rule 含嚴謹原始條款（例如「外殼發燙必須以隔熱物接觸」）-> 放行，不被誤殺
    good_safety = dict(valid_inputs)
    good_safety["safety_rule"] = "外殼發燙必須以隔熱物接觸，嚴禁直接觸碰出風口"
    res_safety = build_candidate_quintuple(**good_safety)
    assert res_safety.is_valid is True
    assert res_safety.slots[EH4_PREDICATE_SAFETY_RULE] == good_safety["safety_rule"]

    # 4. duty_action 含正常動作分工 -> 放行
    good_duty = dict(valid_inputs)
    good_duty["duty_action"] = "我負責在旁看顧，先問過主人才動手"
    res_duty = build_candidate_quintuple(**good_duty)
    assert res_duty.is_valid is True
    assert res_duty.slots[EH4_PREDICATE_DUTY_ACTION] == good_duty["duty_action"]


def test_subject_tail_trim_conservative_policy():
    """驗收項 7：主詞清洗保守策略，歧義殘留尾綴「就」拒收，合法詞彙通過。"""
    # 歧義尾綴：拒收
    ok1, _, err1 = validate_clean_subject("氣炸鍋就")
    assert ok1 is False
    assert "ambiguous trailing particle '就'" in err1

    # 合法名詞：放行
    for term in ("成就", "造就", "遷就", "將就"):
        ok, res, _ = validate_clean_subject(term)
        assert ok is True
        assert res == term

    # 正常實體：放行
    ok3, res3, _ = validate_clean_subject("氣炸鍋")
    assert ok3 is True
    assert res3 == "氣炸鍋"


def test_alignment_with_existing_provider_four_slots_output():
    """驗收項 8：與現有 build_concept_quintuple_slots() 代表性輸出做對照測試。
    證明該代表性輸出與純候選結構驗證相容（不宣稱單例能證明全量 SF-2 語意）。
    """
    user_text = "氣炸鍋就是個插電吹熱風把食物烤熟的箱子，外殼會燙要注意。"
    existing_slots = build_concept_quintuple_slots("agent_rem", "氣炸鍋", user_text)
    assert existing_slots is not None
    assert set(existing_slots.keys()) == set(EH4_DERIVED_PREDICATES)

    # 送入純結構驗證器
    candidate_input = {
        "concept": "氣炸鍋",
        "mental_model": existing_slots[EH4_PREDICATE_MENTAL_MODEL],
        "safety_rule": existing_slots[EH4_PREDICATE_SAFETY_RULE],
        "duty_action": existing_slots[EH4_PREDICATE_DUTY_ACTION],
        "idiolect": existing_slots[EH4_PREDICATE_IDIOLECT],
        "evidence_text": user_text,
        "session_id": "test_alignment_001",
        "source_pair": "bryan:agent_rem",
        "event_time": 1700000000.0,
    }

    res = build_candidate_quintuple(**candidate_input)
    assert res.is_valid is True
    assert res.slots == existing_slots


def test_mechanical_proof_zero_production_add_fact_called(valid_inputs, monkeypatch):
    """驗收項 9：機械證明生產 add_fact 呼叫為 0。若底層被呼叫立即觸發 pytest.fail。"""
    def forbidden_add_fact(*args, **kwargs):
        pytest.fail("CRITICAL VIOLATION: Production GraphStore.add_fact was called by pure candidate builder!")

    monkeypatch.setattr(GraphStore, "add_fact", forbidden_add_fact)

    res = build_candidate_quintuple(**valid_inputs)
    assert res.is_valid is True
    assert len(res.slot_candidates) == 4
